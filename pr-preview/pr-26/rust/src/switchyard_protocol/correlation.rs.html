4//! Where a provider's own request id can be found, and what counts as one.
5//!
6//! WHY THIS IS SHARED
7//!
8//! Two layers need this answer and they must not disagree:
9//!
10//! * the HTTP client, which is the only place a response header still exists and
11//!   which therefore has to decide whether the error body already names a request;
12//! * the server, which reads a failed request's body and reports the id.
13//!
14//! When each carried its own list of locations, the writer checked fewer than the
15//! reader: a body whose only id was in the provider's message prose was treated as
16//! unnamed, so a header id was injected — and the reader, preferring the structured
17//! field, then logged the injected header id instead of the provider's. One
18//! definition, in the crate both depend on, removes that class of divergence.
19//!
20//! WHAT A PROVIDER ID IS
21//!
22//! Correlation only works if the value is exactly the string the provider logged.
23//! A value that has been trimmed of awkward characters, truncated, or otherwise
24//! "repaired" is a different string that resolves to no request, and it is worse
25//! than reporting nothing because it looks like evidence. So every entry point here
26//! validates and refuses; none of them repairs.
27
28use serde_json::Value;
29
30/// The longest identifier any provider is known to issue. A longer value is not a
31/// truncated identifier, it is something else.
32pub const MAX_REQUEST_ID_LEN: usize = 128;
33
34/// Structured field names a provider may use for its request id.
35const REQUEST_ID_FIELDS: &[&str] = &["request_id", "requestId", "requestID"];
36
37/// Accepts a header value only if it is exactly a provider identifier.
38///
39/// Surrounding optional whitespace is trimmed because that is transport framing, not
40/// identity. Every remaining byte must be an identifier character, and the value must
41/// fit in [`MAX_REQUEST_ID_LEN`]. Anything else — an embedded quote, a backslash, a
42/// comma-separated list, interior whitespace, control characters, an empty value, an
43/// over-long value — yields `None` rather than a rewritten identifier.
44pub fn validated_request_id(raw: &str) -> Option<String> {
45    let trimmed = raw.trim_matches(|c: char| c == ' ' || c == '\t');
46    let identifier_shaped = !trimmed.is_empty()
47        && trimmed
48            .chars()
49            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.' | ':'));
50    // Length is compared in bytes, which equals characters here because every byte
51    // above is ASCII.
52    (identifier_shaped && trimmed.len() <= MAX_REQUEST_ID_LEN).then(|| trimmed.to_string())
53}
54
55/// Whether the document already carries a provider-authored request id.
56///
57/// Checks every location [`provider_request_id`] reads, so the layer that decides
58/// whether to supplement a body and the layer that reports from it cannot disagree
59/// about what "already named" means.
60pub fn names_a_request(document: &Value) -> bool {
61    provider_request_id(document).is_some()
62}
63
64/// The provider's request id, from every location providers are known to use.
65///
66/// Priority is provider-authored evidence, most structured first:
67///
68/// 1. a structured field nested under `error` (the OpenAI shape);
69/// 2. a structured field at the top level (the Anthropic shape);
70/// 3. the provider's own message prose, e.g. DeepSeek's `(request_id: 5502af93-…)`.
71///
72/// A bare top-level `id` is deliberately never read: that is the response *object*
73/// id (`resp_…`, `chatcmpl_…`), and reporting it would put a value beside the
74/// provider's logs that does not exist there.
75pub fn provider_request_id(document: &Value) -> Option<String> {
76    if let Some(id) = document.get("error").and_then(structured_request_id) {
77        return Some(id);
78    }
79    if let Some(id) = structured_request_id(document) {
80        return Some(id);
81    }
82    document
83        .get("error")
84        .and_then(|error| error.get("message"))
85        .and_then(Value::as_str)
86        .and_then(request_id_in_text)
87}
88
89/// A validated structured field on one object, if it carries one.
90fn structured_request_id(object: &Value) -> Option<String> {
91    REQUEST_ID_FIELDS
92        .iter()
93        .find_map(|field| object.get(*field).and_then(Value::as_str))
94        .and_then(validated_request_id)
95}
96
97/// The shortest prose token that can be a request id rather than a word.
98///
99/// Prose carries no delimiter that separates an identifier from the sentence around
100/// it, so shape alone is not enough: "the request id header is missing" would yield
101/// `header`. Two guards, both of which a real identifier from every provider on
102/// record passes:
103///
104/// * at least [`MIN_PROSE_ID_LEN`] characters - an eight-character English word is
105///   plausible, a shorter one is not an identifier;
106/// * at least one ASCII digit - provider ids are uuid-, hex- or counter-shaped, and
107///   the providers that write them into prose (DeepSeek) always include digits.
108///
109/// Neither guard applies to a structured field or a header, where the value's
110/// position already says what it is.
111const MIN_PROSE_ID_LEN: usize = 8;
112
113/// Finds a provider request id written into prose.
114///
115/// Providers that report it this way give no structured field, so the message text is
116/// the only place it exists. The token must look like an identifier and be bounded in
117/// length, so ordinary words cannot match; the surrounding message is never returned,
118/// copied, or logged.
119pub fn request_id_in_text(text: &str) -> Option<String> {
120    // Bound the scan: an error message is small, and a huge body is not worth it.
121    const SCAN_LIMIT: usize = 64 * 1024;
122    // Truncate on a CHARACTER boundary: a body may hold multi-byte UTF-8, and slicing
123    // mid-character would panic on the error path.
124    let limit = text.len().min(SCAN_LIMIT);
125    let end = (0..=limit)
126        .rev()
127        .find(|offset| text.is_char_boundary(*offset))
128        .unwrap_or(0);
129    let text = &text[..end];
130    let lower = text.to_ascii_lowercase();
131    for marker in ["request_id", "request id", "request-id"] {
132        let mut from = 0;
133        while let Some(offset) = lower[from..].find(marker) {
134            let rest = text[from + offset + marker.len()..]
135                .trim_start_matches([' ', '\t', ':', '=', '"', '\'', '(', '#']);
136            let token: String = rest
137                .chars()
138                .take_while(|c| c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.' | ':'))
139                .collect();
140            let looks_like_an_id =
141                token.len() >= MIN_PROSE_ID_LEN && token.chars().any(|c| c.is_ascii_digit());
142            if looks_like_an_id && let Some(id) = validated_request_id(&token) {
143                return Some(id);
144            }
145            from += offset + marker.len();
146        }
147    }
148    None
149}

151#[cfg(test)]
152mod tests {
153    use super::*;
154    use serde_json::json;
155
156    #[test]
157    fn an_identifier_shaped_value_is_preserved_exactly() {
158        for value in [
159            "req_abc123",
160            "5502af93-ffbb-46bc-95ec-6402a807b7e8",
161            "req-1.example_v2:a",
162        ] {
163            assert_eq!(validated_request_id(value).as_deref(), Some(value));
164        }
165        assert_eq!(
166            validated_request_id("  req_padded\t ").as_deref(),
167            Some("req_padded")
168        );
169    }
170
171    #[test]
172    fn a_malformed_value_is_refused_not_repaired() {
173        for value in [
174            r#"req_ok","injected":"yes"#,
175            r#"req_ok\injected"#,
176            "req_a,req_b",
177            "req ok",
178            "(req_ok)",
179            "",
180            "   ",
181        ] {
182            assert_eq!(validated_request_id(value), None, "must refuse {value:?}");
183        }
184    }
185
186    #[test]
187    fn an_over_long_value_is_refused_not_truncated() {
188        let at_limit = "a".repeat(MAX_REQUEST_ID_LEN);
189        assert_eq!(
190            validated_request_id(&at_limit).as_deref(),
191            Some(at_limit.as_str())
192        );
193        for over in ["a".repeat(MAX_REQUEST_ID_LEN + 1), "a".repeat(5_000)] {
194            assert_eq!(validated_request_id(&over), None);
195        }
196    }
197
198    #[test]
199    fn a_structured_field_wins_over_prose() {
200        let document = json!({
201            "error": {
202                "message": "failed (request_id: req_from_prose)",
203                "request_id": "req_from_field"
204            }
205        });
206        assert_eq!(
207            provider_request_id(&document).as_deref(),
208            Some("req_from_field")
209        );
210    }
211
212    #[test]
213    fn a_top_level_field_is_read() {
214        let document = json!({
215            "type": "error",
216            "error": {"message": "x"},
217            "request_id": "req_top"
218        });
219        assert_eq!(provider_request_id(&document).as_deref(), Some("req_top"));
220    }
221
222    // The case that mattered: prose is the ONLY id, and it must be found so a header
223    // cannot be injected over it.
224    #[test]
225    fn prose_is_found_when_no_structured_field_exists() {
226        let document = json!({
227            "error": {
228                "message": "thinking mode failure (request_id: req_body_123)",
229                "type": "upstream_error"
230            }
231        });
232        assert_eq!(
233            provider_request_id(&document).as_deref(),
234            Some("req_body_123")
235        );
236        assert!(names_a_request(&document));
237    }
238
239    // A bare top-level `id` is a response object id, not a request id.
240    #[test]
241    fn a_top_level_object_id_is_not_a_request_id() {
242        let document = json!({"id": "resp_object_id", "error": {"message": "x"}});
243        assert_eq!(provider_request_id(&document), None);
244        assert!(!names_a_request(&document));
245    }
246
247    // Control: prose mentioning the words without an identifier must not match.
248    #[test]
249    fn prose_without_an_identifier_does_not_match() {
250        for message in [
251            "the request id header is missing",
252            "request_id was rejected",
253            "no request-id supplied",
254        ] {
255            let document = json!({"error": {"message": message}});
256            assert_eq!(
257                provider_request_id(&document),
258                None,
259                "must not invent an id from {message:?}"
260            );
261        }
262    }
263
264    #[test]
265    fn a_multibyte_body_does_not_panic() {
266        let padding = "漢".repeat(24_000);
267        let document = json!({"error": {"message": format!("{padding} (request_id: 0123abcd-4567-89ef-0123-456789abcdef)")}});
268        // The id sits past the scan limit, so it is not found — but nothing panics.
269        assert_eq!(provider_request_id(&document), None);
270
271        let document = json!({"error": {"message": format!("failed (request_id: 0123abcd-4567-89ef-0123-456789abcdef) {padding}")}});
272        assert_eq!(
273            provider_request_id(&document).as_deref(),
274            Some("0123abcd-4567-89ef-0123-456789abcdef")
275        );
276    }
