3/// Whether a target's model is asked to reason, and how that reaches the provider.
4///
5/// `reasoning_effort` covers the OpenAI-shaped dialects, where the knob is a value
6/// (`none`, `low`, ... `max`). Providers that accept reasoning but expose it as a
7/// boolean switch have no effort value, so they carry `Enabled`/`Disabled` instead.
8/// The distinction matters because a strict non-reasoning target must be able to say so
9/// even when the caller asked for reasoning: this is applied after the caller's fields
10/// and after any `extra_body` default, so the target's choice wins.
11#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
12#[serde(rename_all = "snake_case")]
13pub enum ReasoningPolicy {
14    /// Leave reasoning to the provider's own default.
15    Unspecified,
16    /// Ask the provider not to reason (`reasoning.enabled = false`, or
17    /// `chat_template_kwargs.enable_thinking = false` where that is the dialect).
18    Disabled,
19    /// Ask the provider to reason.
20    Enabled,
21}
22
23impl ReasoningPolicy {
24    /// Whether this policy names a concrete choice. `Unspecified` defers to the
25    /// provider and therefore must not clear fields the caller or target already set.
26    pub fn is_specified(self) -> bool {
27        !matches!(self, Self::Unspecified)
28    }
29
30    /// The provider-shaped body fragment for a dialect, or `None` when the dialect has
31    /// no representation and the policy must fail closed rather than send a request
32    /// that silently reasons.
33    pub fn wire_body(self, dialect: ReasoningDialect) -> Option<serde_json::Value> {
34        use serde_json::json;
35        match dialect {
36            ReasoningDialect::OpenAiEffort => match self {
37                Self::Unspecified => None,
38                Self::Disabled => Some(json!({"reasoning": {"effort": "none"}})),
39                Self::Enabled => None,
40            },
41            ReasoningDialect::OpenRouterEnabled => match self {
42                Self::Unspecified => None,
43                Self::Disabled => Some(json!({"reasoning": {"enabled": false}})),
44                Self::Enabled => Some(json!({"reasoning": {"enabled": true}})),
45            },
46            ReasoningDialect::LlamaCppEnableThinking => match self {
47                Self::Unspecified => None,
48                Self::Disabled => {
49                    Some(json!({"chat_template_kwargs": {"enable_thinking": false}}))
50                }
51                Self::Enabled => Some(json!({"chat_template_kwargs": {"enable_thinking": true}})),
52            },
53        }
54    }
55}
56
57/// How a provider expects reasoning control on the wire.
58///
59/// This is a property of the provider, not of the request, so it belongs on the target
60/// rather than on the transport: two targets behind one transport routinely use
61/// different dialects.
62#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
63#[serde(rename_all = "snake_case")]
64pub enum ReasoningDialect {
65    /// `reasoning.effort` with graded values, including `none`.
66    #[default]
67    #[serde(rename = "openai_effort")]
68    OpenAiEffort,
69    /// `reasoning.enabled`, a boolean switch with no graded values.
70    #[serde(rename = "openrouter_enabled")]
71    OpenRouterEnabled,
72    /// `chat_template_kwargs.enable_thinking`, as llama.cpp servers accept it.
73    #[serde(rename = "llama_cpp_enable_thinking")]
74    LlamaCppEnableThinking,
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80    use serde_json::json;
81
82    #[test]
83    fn unspecified_defers_and_never_clears() {
84        assert!(!ReasoningPolicy::Unspecified.is_specified());
85        assert!(ReasoningPolicy::Disabled.is_specified());
86        assert!(ReasoningPolicy::Enabled.is_specified());
87        assert_eq!(
88            ReasoningPolicy::Unspecified.wire_body(ReasoningDialect::OpenRouterEnabled),
89            None
90        );
91    }
92
93    #[test]
94    fn openai_effort_cannot_express_enabled() {
95        // `none` is the only off value on the OpenAI dialect. There is no way to name
96        // "on" without an effort value, so `Enabled` has no fragment and the target keeps
97        // the caller's effort rather than inventing one.
98        assert_eq!(
99            ReasoningPolicy::Enabled.wire_body(ReasoningDialect::OpenAiEffort),
100            None
101        );
102        assert_eq!(
103            ReasoningPolicy::Disabled.wire_body(ReasoningDialect::OpenAiEffort),
104            Some(json!({"reasoning": {"effort": "none"}}))
105        );
106    }
107
108    #[test]
109    fn boolean_dialects_round_trip_both_directions() {
110        for dialect in [
111            ReasoningDialect::OpenRouterEnabled,
112            ReasoningDialect::LlamaCppEnableThinking,
113        ] {
114            assert!(ReasoningPolicy::Disabled
115                .wire_body(dialect)
116                .is_some_and(|body| body.to_string().contains("false")));
117            assert!(ReasoningPolicy::Enabled
118                .wire_body(dialect)
119                .is_some_and(|body| body.to_string().contains("true")));
120        }
121    }
122
123    #[test]
124    fn policy_round_trips_through_the_config_schema() {
125        for (json_text, expected) in [
126            ("\"disabled\"", ReasoningPolicy::Disabled),
127            ("\"enabled\"", ReasoningPolicy::Enabled),
128            ("\"unspecified\"", ReasoningPolicy::Unspecified),
129        ] {
130            #[derive(Deserialize)]
131            struct Holder {
132                reasoning_policy: ReasoningPolicy,
133            }
134            let holder: Holder =
135                serde_json::from_str(&format!("{{\"reasoning_policy\": {json_text}}}"))
136                    .expect("policy parses");
137            assert_eq!(holder.reasoning_policy, expected);
138        }
139    }
140
141    #[test]
142    fn an_unknown_policy_name_is_rejected_rather_than_defaulted() {
143        // A typo must fail closed at load time, not silently leave reasoning unspecified.
144        #[derive(Deserialize)]
145        #[allow(dead_code)]
146        struct Holder {
147            reasoning_policy: ReasoningPolicy,
148        }
149        assert!(
150            serde_json::from_str::<Holder>(r#"{"reasoning_policy": "disbled"}"#).is_err(),
151            "unknown policy name must not parse"
152        );
153    }
